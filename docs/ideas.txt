=============== Titles and frequencies ==================
titles = {}
for name in df["Name"]:
    # Cut the family name
    stripped_string = name[name.find(", ") + 2:]

    # Append the title
    title = stripped_string[:stripped_string.find(" ")]
    if title in titles:
        titles[title] += 1
    else:
        titles[title] = 1

print("Titles with their respective frequencies:\n", titles)
